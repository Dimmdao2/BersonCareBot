type Row = Record<string, unknown>;
const q = async (t: string, v: unknown[] = []) => (await c.query(t, v)).rows;

async function mkUser(opts: {
  display: string; last?: string | null; first?: string | null; patr?: string | null; created?: string;
}) {
  const r = await q(
    `INSERT INTO platform_users (display_name, role, first_name, last_name, patronymic, created_at)
     VALUES ($1,'client',$2,$3,$4,COALESCE($5::timestamptz, now())) RETURNING id`,
    [opts.display, opts.first ?? null, opts.last ?? null, opts.patr ?? null, opts.created ?? null],
  );
  const id = r[0]!.id as string;
  await q(
    `INSERT INTO user_identity (platform_user_id, first_name, last_name, patronymic, display_name)
     VALUES ($1,$2,$3,$4,$5)`,
    [id, opts.first ?? null, opts.last ?? null, opts.patr ?? null, opts.display],
  );
  return id;
}

async function summary(id: string) {
  const r = await q(
    `SELECT id, display_name, first_name, last_name, patronymic, created_at FROM platform_users WHERE id=$1`, [id],
  );
  const x = r[0]!;
  return {
    id: x.id as string,
    displayName: x.display_name as string,
    firstName: x.first_name as string | null,
    lastName: x.last_name as string | null,
    patronymic: x.patronymic as string | null,
    createdAt: x.created_at as Date,
  };
}

async function show(label: string, id: string) {
  const pu = (await q(`SELECT display_name, first_name, last_name, patronymic, merged_into_id FROM platform_users WHERE id=$1`, [id]))[0];
  const ui = (await q(`SELECT display_name, first_name, last_name, patronymic FROM user_identity WHERE platform_user_id=$1`, [id]))[0];
  const ch = await q(`SELECT channel_code, external_id FROM user_channel_bindings WHERE user_id=$1 ORDER BY external_id`, [id]);
  const ct = await q(`SELECT contact_kind, value_normalized, is_primary, confirmed_at IS NOT NULL AS confirmed FROM user_contacts WHERE platform_user_id=$1 ORDER BY contact_kind, value_normalized`, [id]);
  console.log(`--- ${label} platform_users:`, JSON.stringify(pu));
  console.log(`--- ${label} user_identity :`, JSON.stringify(ui));
  console.log(`--- ${label} channels      :`, JSON.stringify(ch));
  console.log(`--- ${label} contacts      :`, JSON.stringify(ct));
}

async function scenario(name: string, fn: () => Promise<void>) {
  await c.query('BEGIN');
  console.log(`\n================ ${name} ================`);
  try { await fn(); } catch (e) { console.log('THREW:', (e as Error).message); }
  await c.query('ROLLBACK');
}

// A. legacy display-only, different names: must be a human conflict; chosen variant lands everywhere
await scenario('A: legacy display-only conflict', async () => {
  const t = await mkUser({ display: 'Иванов Иван Петрович', created: '2026-01-01' });
  const d = await mkUser({ display: 'Сидорова Анна Сергеевна', created: '2025-01-01' });
  await q(`INSERT INTO user_channel_bindings (user_id, channel_code, external_id) VALUES ($1,'telegram','tg-initiator'),($2,'telegram','tg-real-owner'),($2,'max','max-real-owner')`, [t, d]);
  await q(`INSERT INTO user_contacts (platform_user_id, contact_kind, value_normalized, is_primary, confirmed_at, source_origin)
           VALUES ($1,'phone','+79990000001',true,now(),'direct'),($2,'email','old-owner@example.test',true,now(),'direct')`, [t, d]);
  const prompt = createHumanMergePrompt(await summary(t), await summary(d), d);
  console.log('conflicts =', JSON.stringify(prompt.conflicts));
  try {
    await mergePlatformUsersInTransaction(c, t, d, 'phone_bind', { humanDecision: createHumanMergeDecision(prompt, {}) });
    console.log('!! merged with NO choice');
  } catch (e) { console.log('no-choice merge refused:', (e as Error).message); }
  await mergePlatformUsersInTransaction(c, t, d, 'phone_bind', {
    humanDecision: createHumanMergeDecision(prompt, { display_name: { source: 'duplicate' } }),
  });
  await show('target', t);
  await show('dup   ', d);
});
